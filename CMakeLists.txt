cmake_minimum_required(VERSION 3.20)
project(leetcode CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(SANITIZE "Build with ASan/UBSan" OFF)

include(FetchContent)
set(INSTALL_GTEST OFF)
FetchContent_Declare(googletest
  URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip)
FetchContent_MakeAvailable(googletest)

enable_testing()
include(GoogleTest)

foreach(dir leetcode epi embedded trivia)
  file(GLOB SRCS CONFIGURE_DEPENDS ${dir}/*.cpp)
  if (SRCS)
    add_executable(${dir}_tests ${SRCS})
    target_include_directories(${dir}_tests PRIVATE ${PROJECT_SOURCE_DIR}/include/${dir})
    target_compile_options(${dir}_tests PRIVATE -Wall -Wextra -Wpedantic)
    if (SANITIZE)
      target_compile_options(${dir}_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${dir}_tests PRIVATE -fsanitize=address,undefined)
    endif()
    target_link_libraries(${dir}_tests PRIVATE GTest::gmock_main)
    gtest_discover_tests(${dir}_tests)
  endif()
endforeach()
